  const p=window.__bikeGetStatus?.()?.pageParams||{};
  for(const k of Object.keys(display))if(typeof p[k]==='boolean')display[k]=p[k];
 };
 const saveCamera=()=>{if(tabState==='camera'&&!applying){cameraView=window.__bikeGetStatus?.()?.view||cameraView;}};
 function setup(){
  const root=document.getElementById('reactControls'), lab=document.getElementById('machineLab');
  const sections=root?.querySelectorAll(':scope > div > section.section');
  if(!root||!lab||!sections||sections.length<5){requestAnimationFrame(setup);return;}
  if(root.dataset.v344Ready)return;
  root.dataset.v344Ready='1';
  const nav=document.createElement('nav');nav.className='compactNav';nav.setAttribute('aria-label','操作分区');
  nav.innerHTML='<div class="compactNavHead"><strong>教学操作</strong><span>独立状态</span></div><div class="compactTabs" role="tablist"></div><div class="compactTip" aria-live="polite"></div>';
  root.insertBefore(nav,root.firstChild);
  const btnwrap=nav.querySelector('.compactTabs');
  tabs.forEach(([key,label])=>{
   const b=document.createElement('button');b.type='button';b.textContent=label;b.dataset.tab=key;b.setAttribute('role','tab');
   b.addEventListener('click',()=>switchTab(key));btnwrap.appendChild(b);
  });
  const diagram=lab.querySelector('.machineLabDiagram'), caution=lab.querySelector('.machineLabCaution');
  if(diagram){const details=document.createElement('details');details.className='compactMachineDiagram';
   const sum=document.createElement('summary');sum.textContent='展开原理图和教学说明';details.appendChild(sum);
   details.appendChild(diagram);if(caution)details.appendChild(caution);lab.appendChild(details);
  }
  window.addEventListener('bikeUI',()=>{saveManual();saveDisplay();saveCamera();});
  // Direct slider/change events are captured after React reconciles the new state.
  root.addEventListener('change',()=>{if(tabState==='adjust')queueMicrotask(saveManual);});
  function switchTab(tab){
   if(!tips[tab])return;
   if(tabState===tab)return;
   saveManual();saveDisplay();saveCamera();
   applying=true;
   // Stop the previous demo BEFORE adjusting any new-page parameters.
   window.__machineLab?.stop();
   if(window.__bikeStopMotion)window.__bikeStopMotion({clearAngles:false});
   else window.__bikeSetParams?.({spin:false});
   tabState=tab;
   root.dataset.compactTab=tab;
   for(const b of btnwrap.querySelectorAll('button')){
    const active=b.dataset.tab===tab;
    b.classList.toggle('active',active);b.setAttribute('aria-selected',active?'true':'false');b.tabIndex=active?0:-1;
   }
   nav.querySelector('.compactTip').textContent=tips[tab];
   if(tab==='demo'){
     window.__bikeSetParams?.({spin:false,brake:0,steer:0});
     window.__machineLab?.play?.();
   }else if(tab==='adjust'){
     // This is exactly what the manual controls should display and execute.
     window.__bikeSetParams?.({...display,...manual});
   }else{
     // Camera/display pages own the display toggles; an old demo must not overwrite them.
     window.__bikeSetParams?.({...display,spin:false,brake:0,steer:0});
     if(tab==='camera')window.__bikeSetView?.(cameraView);
   }
   root.scrollTop=0;
   applying=false;
   window.dispatchEvent(new CustomEvent('bikeTabChange',{detail:{activeTab:tab}}));
  }
  // Preserve default Demo tab and its actively selected demonstration.
  tabState='';switchTab('demo');
  window.__bikeCompactUI={get activeTab(){return tabState},setTab:switchTab,tabs:tabs.map(([id])=>id),get manual(){return {...manual}},get display(){return {...display}},get cameraView(){return cameraView},get state(){return {sections:sections.length,lab:!!document.getElementById('machineLab'),playing:!!window.__machineLab?.playing,collapsedDiagram:!lab.querySelector('details')?.open}}};
 }
 if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',setup);else setup();
})();</script>
